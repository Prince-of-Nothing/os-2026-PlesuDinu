== Part 2: shell expansion ==
$ echo *.c
a.c b.c
$ echo file?.txt
file1.txt file2.txt
$ echo [a-b]*
a.c alfa beta.txt b.c
$ echo *.xyz
*.xyz
$ ls alfa beta.txt
ls: cannot access 'alfa': No such file or directory
ls: cannot access 'beta.txt': No such file or directory
$ ls "alfa beta.txt"
alfa beta.txt
$ echo '$HOME' "$HOME" \$HOME
$HOME /home/user $HOME
$ echo {1..5} ; echo $((7 * 6))
1 2 3 4 5
42
$ echo "today is $(date +%A), I am $(whoami)"
today is Friday, I am user
$ touch log{1..3}.txt && ls log*
log1.txt
log2.txt
log3.txt
